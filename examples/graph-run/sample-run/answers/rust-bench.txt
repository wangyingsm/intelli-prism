// Add to Cargo.toml:
//   [dev-dependencies]
//   criterion = { version = "0.5", features = ["html_reports"] }
//
//   [[bench]]
//   name = "merge_sort_bench"
//   harness = false

use criterion::{black_box, criterion_group, criterion_main, Criterion, Throughput};

pub fn merge_sort(mut v: Vec<i64>) -> Vec<i64> {
    let len = v.len();
    if len <= 1 {
        return v;
    }
    let mid = len / 2;
    let right = v.split_off(mid);
    let left = merge_sort(v);
    let right = merge_sort(right);

    let mut merged = Vec::with_capacity(len);
    let mut i = 0;
    let mut j = 0;
    while i < left.len() && j < right.len() {
        if left[i] <= right[j] {
            merged.push(left[i]);
            i += 1;
        } else {
            merged.push(right[j]);
            j += 1;
        }
    }
    merged.extend_from_slice(&left[i..]);
    merged.extend_from_slice(&right[j..]);
    merged
}

/// Small deterministic xorshift PRNG so the input is reproducible without extra deps.
struct Rng(u64);

impl Rng {
    fn new(seed: u64) -> Self {
        Rng(seed.max(1))
    }
    fn next_u64(&mut self) -> u64 {
        let mut x = self.0;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.0 = x;
        x
    }
    fn next_i64(&mut self) -> i64 {
        self.next_u64() as i64
    }
}

fn bench_merge_sort(c: &mut Criterion) {
    const N: usize = 10_000;

    let mut rng = Rng::new(0xDEADBEEFCAFEBABE);
    let data: Vec<i64> = (0..N).map(|_| rng.next_i64()).collect();

    let mut group = c.benchmark_group("merge_sort");
    group.throughput(Throughput::Elements(N as u64));

    group.bench_function("random_i64_10k", |b| {
        b.iter(|| {
            // black_box the input clone so LLVM cannot hoist/eliminate the sort.
            let input = black_box(data.clone());
            black_box(merge_sort(input));
        });
    });

    group.finish();
}

criterion_group!(benches, bench_merge_sort);
criterion_main!(benches);
